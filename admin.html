/* ===================================================== CREATE VIDEO CARD ===================================================== */ function createVideoCard( video ){ const card = document.createElement( "div" ); card.className = "video-card"; const thumbnailURL = getPublicURL( video.thumbnail_url, "thumbnails" ); const title = video.title || "Untitled"; const category = video.category || "General"; const description = video.description || "No description"; const uploaded = video.created_at ? new Date( video.created_at ).toLocaleDateString() : ""; /* LIKES We will read the likes count that loadVideos() provides for this video. */ const likes = Number( video.likes || 0 ); card.innerHTML = `
${ thumbnailURL ? ` ` : `
🎬
` }
${escapeHTML(category)}

${escapeHTML(title)}

${escapeHTML(description)}

Uploaded: ${escapeHTML(uploaded)}

Views: ${Number( video.views || 0 ).toLocaleString()}

❤️ Likes: ${likes.toLocaleString()}

`; document .getElementById("videoGrid") .appendChild(card); }
